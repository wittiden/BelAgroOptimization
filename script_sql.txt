-- ==========================================================
DO $$
DECLARE
    tables text[] := ARRAY[
        'scenarios', 'crops', 'fields', 'crop_data', 'field_data',
        'weather_data', 'livestock_data', 'feed_data', 'feed_outputs',
        'optimization_results', 'crop_allocations', 'livestock_allocations',
        'feed_allocations', 'monthly_operations'
    ];
    tablename text;
    id_column text;
BEGIN
    FOREACH tablename IN ARRAY tables
    LOOP
        CASE tablename
            WHEN 'scenarios' THEN id_column := 'scenario_id';
            WHEN 'crops' THEN id_column := 'crop_id';
            WHEN 'fields' THEN id_column := 'field_id';
            WHEN 'crop_data' THEN id_column := 'crop_data_id';
            WHEN 'field_data' THEN id_column := 'field_data_id';
            WHEN 'weather_data' THEN id_column := 'weather_data_id';
            WHEN 'livestock_data' THEN id_column := 'livestock_data_id';
            WHEN 'feed_data' THEN id_column := 'feed_data_id';
            WHEN 'feed_outputs' THEN id_column := 'feed_output_id';
            WHEN 'optimization_results' THEN id_column := 'result_id';
            WHEN 'crop_allocations' THEN id_column := 'crop_allocation_id';
            WHEN 'livestock_allocations' THEN id_column := 'livestock_allocation_id';
            WHEN 'feed_allocations' THEN id_column := 'feed_allocation_id';
            WHEN 'monthly_operations' THEN id_column := 'monthly_operation_id';
        END CASE;

        EXECUTE format('ALTER TABLE %I ALTER COLUMN %I SET DEFAULT gen_random_uuid();', tablename, id_column);
    END LOOP;
END $$;

-- ==========================================================
-- 10. ПРОВЕРКА РЕНТАБЕЛЬНОСТИ ЖИВОТНОВОДСТВА
-- ==========================================================
DO $$
DECLARE
    var_scenario_id UUID;
    cow_profit numeric;
    cattle_profit numeric;
    pig_profit numeric;
BEGIN
    SELECT scenario_id INTO var_scenario_id FROM scenarios WHERE name = 'Базовый сценарий 2024-2026' LIMIT 1;

    -- Расчет годовой прибыли на 1 голову
    SELECT
        ROUND((base_yield_kg * price_byn_per_kg) - (cost_summer_byn + cost_winter_byn + energy_cost_winter_byn), 0)
    INTO cow_profit
    FROM livestock_data
    WHERE scenario_id = var_scenario_id AND animal_type = 'cow';

    SELECT
        ROUND((base_yield_kg * price_byn_per_kg) - (cost_summer_byn + cost_winter_byn + energy_cost_winter_byn), 0)
    INTO cattle_profit
    FROM livestock_data
    WHERE scenario_id = var_scenario_id AND animal_type = 'cattle';

    SELECT
        ROUND((base_yield_kg * price_byn_per_kg) - (cost_summer_byn + cost_winter_byn + energy_cost_winter_byn), 0)
    INTO pig_profit
    FROM livestock_data
    WHERE scenario_id = var_scenario_id AND animal_type = 'pig';

    RAISE NOTICE '==========================================';
    RAISE NOTICE '💰 РЕНТАБЕЛЬНОСТЬ ЖИВОТНОВОДСТВА:';
    RAISE NOTICE '   Корова: прибыль % BYN/год', cow_profit;
    RAISE NOTICE '   КРС:    прибыль % BYN/год', cattle_profit;
    RAISE NOTICE '   Свинья: прибыль % BYN/год', pig_profit;
    RAISE NOTICE '==========================================';
END $$;

UNION ALL

UNION ALL

UNION ALL

UNION ALL

UNION ALL
